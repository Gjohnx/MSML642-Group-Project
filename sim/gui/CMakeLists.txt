cmake_minimum_required(VERSION 3.17)
project(ChaseButton LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(gz-gui8 REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(ChaseButton SHARED ChaseButton.hh ChaseButton.cc ChaseButton.qrc)
target_link_libraries(ChaseButton PRIVATE gz-gui8::gz-gui8
    gz-transport13::gz-transport13 gz-msgs10::gz-msgs10 gz-plugin2::register)
target_link_options(ChaseButton PRIVATE -Wl,--no-undefined)
